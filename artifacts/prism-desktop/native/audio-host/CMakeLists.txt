cmake_minimum_required(VERSION 3.25)
project(prism_audio_host LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_executable(prism-audio-core-test core-test.cpp)
target_compile_features(prism-audio-core-test PRIVATE cxx_std_17)
add_test(NAME bounded-native-audio-core COMMAND prism-audio-core-test)
option(PRISM_BUILD_AUDIO_DEVICE_HOST "Build the Windows physical-device host" ${WIN32})
option(PRISM_ENABLE_ASIO "Include licensed ASIO support" OFF)
option(PRISM_ASIO_AGREEMENT_EFFECTIVE "An authorized owner verified an effective proprietary agreement" OFF)
set(PRISM_ASIO_SDK_ROOT "" CACHE PATH "Locally supplied licensed ASIO SDK; never downloaded automatically")
if(PRISM_ENABLE_ASIO)
  if(NOT PRISM_ASIO_AGREEMENT_EFFECTIVE OR NOT EXISTS "${PRISM_ASIO_SDK_ROOT}/common/asio.h")
    message(FATAL_ERROR "ASIO requires an effective proprietary agreement and a locally supplied licensed SDK.")
  endif()
endif()
if(NOT PRISM_BUILD_AUDIO_DEVICE_HOST)
  return()
endif()
if(NOT WIN32)
  message(FATAL_ERROR "This device backend targets Windows; core tests remain portable.")
endif()
include(FetchContent)
set(PA_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(PA_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(PA_USE_ASIO ${PRISM_ENABLE_ASIO} CACHE BOOL "" FORCE)
set(PA_USE_WASAPI ON CACHE BOOL "" FORCE)
set(PA_USE_DS OFF CACHE BOOL "" FORCE)
set(PA_USE_WMME OFF CACHE BOOL "" FORCE)
set(PA_USE_WDMKS OFF CACHE BOOL "" FORCE)
set(PA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
if(PRISM_ENABLE_ASIO)
  set(ASIOSDK_ROOT_DIR "${PRISM_ASIO_SDK_ROOT}" CACHE PATH "" FORCE)
endif()
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(portaudio GIT_REPOSITORY https://github.com/PortAudio/portaudio.git
  GIT_TAG 147dd722548358763a8b649b3e4b41dfffbcfbb6)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git
  GIT_TAG 9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03)
FetchContent_MakeAvailable(portaudio json)
add_executable(prism-audio-host main.cpp)
target_compile_features(prism-audio-host PRIVATE cxx_std_17)
target_compile_definitions(prism-audio-host PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN
  PRISM_WITH_ASIO=$<BOOL:${PRISM_ENABLE_ASIO}>)
target_link_libraries(prism-audio-host PRIVATE portaudio_static nlohmann_json::nlohmann_json ole32 uuid)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/notices")
configure_file("${portaudio_SOURCE_DIR}/LICENSE.txt" "${CMAKE_BINARY_DIR}/notices/PortAudio-LICENSE.txt" COPYONLY)
configure_file("${json_SOURCE_DIR}/LICENSE.MIT" "${CMAKE_BINARY_DIR}/notices/JSON-LICENSE.txt" COPYONLY)
